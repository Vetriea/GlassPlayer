/**
 * Real liquid glass surface: samples whatever is drawn behind it through [haze], then adds
 * refraction, depth blur, tint, specular highlight and a rim light edge (iOS Liquid Glass look).
 *
 * The composable that draws the background content must be marked with
 * `Modifier.hazeSource(haze)` — glass elements should be SIBLINGS of that composable,
 * not children, so they never sample themselves.
 */
@Composable
fun LiquidGlass(
    haze: HazeState,
    modifier: Modifier = Modifier,
    corner: RoundedCornerShape = RoundedCornerShape(24.dp),
    tintAlpha: Float = 0.08f,
    base: GlassStyle = GlassStyle.regular,
    content: @Composable BoxScope.() -> Unit,
) {
    // Haze styles compare by identity — remember them or every node re-attaches each recomposition
    val style = remember(corner, tintAlpha, base) {
        base.then {
            tint(Color.White.copy(alpha = tintAlpha))
            shape(corner)
        }
    }
    Box(modifier.hazeGlass(input = HazeInput.Sources(haze), style = style), content = content)
}

/**
 * Lightweight custom seek bar. Replaces Material3's Slider:
 * no deprecated APIs, drag + tap to seek, a thumb that grows while you touch it,
 * and a much thinner, more "glass" look than the M3 default.
 */
@Composable
fun GlassSeekbar(pos: Long, dur: Long, onSeek: (Long) -> Unit, modifier: Modifier = Modifier) {
    var trackW by remember { mutableFloatStateOf(1f) }
    var dragging by remember { mutableStateOf(false) }
    var dragPos by remember { mutableFloatStateOf(0f) }

    val shown = if (dragging) dragPos else pos.toFloat().coerceIn(0f, dur.toFloat())
    val frac = (shown / dur.coerceAtLeast(1L).toFloat()).coerceIn(0f, 1f)
    val thumbScale by animateFloatAsState(if (dragging) 1.3f else 1f, tween(140), label = "thumb")

    fun pxToMs(x: Float) = (x / trackW.coerceAtLeast(1f)) * dur.toFloat()

    Column(modifier.fillMaxWidth()) {
        BoxWithConstraints(
            Modifier
                .fillMaxWidth()
                .height(28.dp)
                .onSizeChanged { trackW = it.width.toFloat() }
                .pointerInput(dur) {
                    detectHorizontalDragGestures(
                        onDragStart = { off ->
                            dragging = true
                            dragPos = pxToMs(off.x)
                        },
                        onDragEnd = {
                            dragging = false
                            onSeek(dragPos.toLong().coerceIn(0L, dur))
                        },
                        onDragCancel = { dragging = false }
                    ) { change, amount ->
                        dragPos = (dragPos + pxToMs(amount)).coerceIn(0f, dur.toFloat())
                        change.consume()
                    }
                }
                .pointerInput(dur) {
                    detectTapGestures { off ->
                        onSeek(pxToMs(off.x).toLong().coerceIn(0L, dur))
                    }
                }
        ) {
            val trackWidth = maxWidth
            // inactive track
            Box(
                Modifier
                    .align(Alignment.CenterStart)
                    .fillMaxWidth()
                    .height(4.dp)
                    .clip(CircleShape)
                    .background(Color.White.copy(alpha = 0.22f))
            )
            // active fill
            Box(
                Modifier
                    .align(Alignment.CenterStart)
                    .fillMaxWidth(frac)
                    .height(4.dp)
                    .clip(CircleShape)
                    .background(Color.White)
            )
            // thumb
            Box(
                Modifier
                    .align(Alignment.CenterStart)
                    .offset(x = (trackWidth - 16.dp) * frac)
                    .size(16.dp)
                    .graphicsLayer { scaleX = thumbScale; scaleY = thumbScale }
                    .clip(CircleShape)
                    .background(Color.White)
            )
        }
        Spacer(Modifier.height(4.dp))
        Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
            Text(fmt(shown.toLong()), color = Color.White.copy(alpha = 0.6f), fontSize = 12.sp)
            Text(fmt(dur), color = Color.White.copy(alpha = 0.6f), fontSize = 12.sp)
        }
    }
}
